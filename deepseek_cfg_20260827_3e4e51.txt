# grub.cfg - GRUB boot configuration
set timeout=5
set default=0

menuentry "MyOS - Single-file C++ x86 Kernel" {
    multiboot /boot/kernel.elf
    boot
}

menuentry "MyOS (verbose)" {
    multiboot /boot/kernel.elf
    boot
}

menuentry "System Information" {
    echo "MyOS - Bare-metal x86"
    echo "Built from single-file C++ kernel"
    echo "Memory: 64MB RAM"
    echo "Features: VGA, Keyboard, Paging, Shell"
    sleep 5
    boot
}
